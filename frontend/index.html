index.html — Integration Snippet 
<!-- 1. Include SDK Script -->
<script src="/sdk/rishiraj-auth.js"></script>

<!-- 2. Initialize Auth Instance -->
<script>
  const auth = new RishirajAuth({
    tenantId: "your-tenant-uuid",
    environment: "DEVELOPMENT",
    onLogin: (user, tokens) => {
      console.log("Welcome back!", user);
    }
  });
</script>

<!-- 3. Trigger Modal Anywhere -->
<button onclick="auth.showLoginModal()">Sign In</button>
